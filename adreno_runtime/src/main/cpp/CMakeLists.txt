cmake_minimum_required(VERSION 3.22.1)
project(localflux_adreno LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SD_WEBP OFF CACHE BOOL "" FORCE)
set(SD_WEBM OFF CACHE BOOL "" FORCE)
set(SD_VULKAN ON CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8.6-a+dotprod+i8mm" CACHE STRING "" FORCE)
set(GGML_CPU_KLEIDIAI ON CACHE BOOL "" FORCE)
set(GGML_CPU_REPACK ON CACHE BOOL "" FORCE)
set(SD_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SD_BUILD_SHARED_GGML_LIB OFF CACHE BOOL "" FORCE)

set(BONSAI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/bonsai-cpp")
if(NOT EXISTS "${BONSAI_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "duration-ai/bonsai-cpp not vendored. Run scripts/vendor-deps.sh first.")
endif()

add_subdirectory("${BONSAI_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/bonsai-cpp")

if(ANDROID AND TARGET ggml-vulkan)
    set(ANDROID_VULKAN_HPP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/vulkan-headers")
    if(EXISTS "${ANDROID_VULKAN_HPP_DIR}/vulkan/vulkan.hpp")
        target_include_directories(ggml-vulkan PRIVATE "${ANDROID_VULKAN_HPP_DIR}")
    else()
        message(FATAL_ERROR "Android Vulkan C++ headers were not staged at ${ANDROID_VULKAN_HPP_DIR}")
    endif()
endif()

add_library(localflux_adreno SHARED adreno-native.cpp)
target_include_directories(localflux_adreno PRIVATE "${BONSAI_DIR}/include")
target_link_libraries(localflux_adreno PRIVATE stable-diffusion android log)
target_link_options(localflux_adreno PRIVATE "-Wl,-z,max-page-size=16384")
